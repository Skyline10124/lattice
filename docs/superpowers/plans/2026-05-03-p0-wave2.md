        return e;
    }
    match self.http_client.get(url).send().await {

            if let Some(content_length) = response.content_length() {

                let truncated = String::from_utf8_lossy(&bytes[..limit]);
                format!(

                    truncated.trim_end(),
                    limit
                )
            } else {
                String::from_utf8_lossy(&bytes).to_string()
            }
        }
        Err(e) => ToolError::HttpError(e.to_string()).to_string(),

fn verify_config_integrity(path: &Path, content: &str) -> Result<(), Box<dyn std::error::Error>> {
    let hash_path = path.with_extension("toml.sha256");
    if !hash_path.exists() {
        if std::env::var("LATTICE_STRICT_CONFIG").unwrap_or_default() == "1" {
            return Err(format!(
                "Config integrity hash not found at {}; set LATTICE_STRICT_CONFIG=0 to allow unverified configs",
                hash_path.display()
            )
            .into());
        }
        return Ok(());
    }
    let expected = std::fs::read_to_string(&hash_path)?.trim().to_string();
    let actual = format!("{:x}", sha2::Sha256::digest(content.as_bytes()));
    if actual != expected {
        return Err(format!(
            "Config integrity mismatch: {} (expected sha256 {}, got {})",
            path.display(),
            expected,
            actual
        )
        .into());
    }
    Ok(())
}
```

fn looks_like_error(content: &str) -> bool {
    let lower = content.trim().to_ascii_lowercase();
    let trimmed = content.trim();

    // Original prefixes (case-sensitive for Error:, error: etc.)
    trimmed.starts_with("Error:")
        || trimmed.starts_with("error:")
        || trimmed.starts_with("Provider error:")
        || trimmed.starts_with("API error:")
        || trimmed.starts_with("Authentication error:")
        || trimmed.starts_with("Rate limit")
        || trimmed.starts_with("401")
        || trimmed.starts_with("403")
        || trimmed.starts_with("429")
        || trimmed.starts_with("500")
        // Provider error patterns (case-insensitive)
        || lower.starts_with("rate limit")
        || lower.starts_with("authentication")
        || lower.starts_with("api key")
        || lower.starts_with("invalid request")
        || lower.starts_with("server error")
        || lower.starts_with("service unavailable")
        // Transport error patterns (case-insensitive)
        || lower.starts_with("timeout")
        || lower.starts_with("connection refused")
        || lower.starts_with("connection reset")
        || lower.starts_with("dns")
        || lower.starts_with("tls")
        || lower.starts_with("certificate")
}

use crate::tools::ToolRegistry;

pub fn assemble_agent(
    profile: &AgentProfile,
    resolved: lattice_core::ResolvedModel,
    memory: Option<Arc<dyn lattice_agent::memory::Memory>>,
    tool_registry: Option<Arc<ToolRegistry>>,
) -> AgentRunner {
    let mut agent = lattice_agent::Agent::new(resolved);

    if let Some(prompt) = profile.system_prompt() {
        agent.set_system_prompt(&prompt);
    }

    let default_tools = lattice_agent::default_tool_definitions();
    let enabled_names: Vec<String> = profile
        .tools
        .enabled
        .iter()
        .map(|s| s.to_string())
        .collect();

    let enabled_tools: Vec<lattice_core::types::ToolDefinition> = if enabled_names.is_empty() {
        default_tools
    } else {
        default_tools
            .into_iter()
            .filter(|t| enabled_names.contains(&t.name))
            .collect()
    };

    // Merge with registry tools if present
    let (final_tools, executor) = if let Some(reg) = &tool_registry {
        if !reg.is_empty() {
            let registry_defs: Vec<lattice_core::types::ToolDefinition> = reg
                .definitions()
                .into_iter()
                .filter(|d| !enabled_tools.iter().any(|et| et.name == d.name))
                .collect();
            let merged = enabled_tools.into_iter().chain(registry_defs).collect();
            let exec = lattice_agent::DefaultToolExecutor::new_with_registry(
                ".",
                Some(reg.clone_as_registry_access()),
            );
            (merged, exec)
        } else {
            let exec = lattice_agent::DefaultToolExecutor::new(".");
            (enabled_tools, exec)
        }
    } else {
        let exec = lattice_agent::DefaultToolExecutor::new(".");
        (enabled_tools, exec)
    };

    if !final_tools.is_empty() {
        agent = agent.with_tools(final_tools);
        agent = agent.with_tool_executor(Box::new(executor));
    }

    if let Some(ref mem) = memory {
        agent = agent.with_memory(mem.clone_box());
    }

    let runner = AgentRunner::from_profile(profile.clone(), agent);

    if let Some(ref mem) = memory {
        runner.with_memory(Arc::clone(mem))
    } else {
        runner
    }
}
```

fn run_async<F, T>(f: F) -> T
where
    F: std::future::Future<Output = T>,
{
    if let Ok(handle) = tokio::runtime::Handle::try_current() {
        // Already in a multi-threaded runtime — block_in_place is safe
        tokio::task::block_in_place(|| handle.block_on(f))
    } else {

        let rt = tokio::runtime::Runtime::new().expect("temp runtime");

    }
}
